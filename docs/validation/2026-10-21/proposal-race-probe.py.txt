 def send(obj): p.stdin.write(json.dumps(obj)+'\n');p.stdin.flush()
 def recv(): return json.loads(p.stdout.readline())
 send({'jsonrpc':'2.0','id':1,'method':'initialize','params':{'protocolVersion':'2024-11-05','capabilities':{},'clientInfo':{'name':'race-fixture','version':'1'}}});recv()
